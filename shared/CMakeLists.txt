add_library(xen_engine STATIC src/engine.cpp src/serialize.cpp)
target_link_libraries(xen_engine PUBLIC xen)
target_include_directories(xen_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(xen_engine PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(xen_engine PRIVATE /W3)
else()
    target_compile_options(xen_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(xen::engine ALIAS xen_engine)
